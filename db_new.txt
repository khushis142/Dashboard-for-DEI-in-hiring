CREATE DATABASE ALWAYS_FIRST;
USE ALWAYS_FIRST;

CREATE TABLE EMPLOYEE(
EMPLOYEE_ID INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
FNAME VARCHAR(20) NOT NULL ,
LNAME VARCHAR(20) NOT NULL,
GENDER CHAR(1) NOT NULL ,
CHECK (GENDER IN ('M', 'F')),
DEPARTMENT_ID INT NOT NULL,
BUSINESS_UNIT VARCHAR(20),
/*CHECK (BUSINESS_UNIT IN('MANUFACTURING', 'HEALTHCARE', 'BANKING', 'LOGISTICS')),*/
SALARY INT NOT NULL,
HIRING_DATE DATE NOT NULL,
POSITION VARCHAR(20) NOT NULL,
CANDIDATE_ID INT NOT NULL,
STATUS VARCHAR(20) NOT NULL,
CHECK (STATUS IN ('LIVE','EXITED')),
CHECK (POSITION IN('ANALYST', 'ASSOCIATE', 'VP', 'ED', 'MD')),
FOREIGN KEY (DEPARTMENT_ID)REFERENCES DEPARTMENTS(DEPARTMENT_ID),
FOREIGN KEY (CANDIDATE_ID)REFERENCES CANDIDATE(CANDIDATE_ID)
);

CREATE TABLE CANDIDATE (
CANDIDATE_ID INT NOT NULL AUTO_INCREMENT,
CANDIDATE_FNAME VARCHAR(30) NOT NULL,
CANDIDATE_LNAME VARCHAR(30) NOT NULL,
GENDER CHAR(1) NOT NULL ,
CHECK (GENDER IN ('M', 'F')),
DEPARTMENT_ID INT(30) NOT NULL,
APPLIED_POSITION VARCHAR(20) NOT NULL,
CHECK (APPLIED_POSITION IN('ANALYST', 'ASSOCIATE', 'VP', 'ED', 'MD')),
APPLIED_THROUGH VARCHAR(20) NOT NULL,
CHECK (APPLIED_THROUGH IN('VENDOR', 'CAMPUS', 'REFERENCE','WEBSITE')),
STATUS VARCHAR(20) NOT NULL,
CHECK (STATUS IN('STAGE 1', 'STAGE 2', 'HIRED')),
EXPERIENCE_YEARS INT NOT NULL,
VENDOR_ID INT ,
CAMPUS_ID INT,
/*EMP_ID_REFERER,*/
APPLICATION_DATE DATE NOT NULL,
PRIMARY KEY (CANDIDATE_ID),
FOREIGN KEY (VENDOR_ID) REFERENCES VENDORS(VENDOR_ID),
FOREIGN KEY (CAMPUS_ID) REFERENCES CAMPUS(CAMPUS_ID),
FOREIGN KEY (DEPARTMENT_ID) REFERENCES DEPARTMENTS(DEPARTMENT_ID)
);



CREATE TABLE VENDORS(
    VENDOR_ID INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    VENDOR_NAME VARCHAR(30) NOT NULL
    );

CREATE TABLE DEPARTMENTS(
    DEPARTMENT_ID INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    DEPARTMENT_NAME VARCHAR(30) NOT NULL
    );

 

CREATE TABLE CAMPUS(
    CAMPUS_ID INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
    CAMPUS_NAME VARCHAR(30) NOT NULL,
    CAMPUS_TYPE VARCHAR(10) NOT NULL ,
    CHECK (CAMPUS_TYPE IN('BOYS','GIRLS','COED'))    
    );
    
CREATE TABLE OPEN_ROLES(
JOB_ID INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
DEPARTMENT_ID INT NOT NULL,
POSITION VARCHAR(20) NOT NULL,
CHECK (POSITION IN('ANALYST', 'ASSOCIATE', 'VP', 'ED', 'MD')), 
BUSINESS_UNIT VARCHAR(20),
CREATION_DATE DATE NOT NULL,
FOREIGN KEY (DEPARTMENT_ID)REFERENCES DEPARTMENTS(DEPARTMENT_ID)

);

CREATE TABLE PERFORMANCE(
EMPLOYEE_ID INT NOT NULL,
SCORE_YEAR YEAR,
SCORE INT,
CHECK (SCORE >= 0 AND SCORE <= 5),
FOREIGN KEY (EMPLOYEE_ID)REFERENCES EMPLOYEE(EMPLOYEE_ID)
);

CREATE TABLE RECOMMENDED_VENDOR (
  RECOMMENDATION_ID INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
  VENDOR_NAME VARCHAR(30) NOT NULL,
  VENDOR_LOCATION VARCHAR(50) NOT NULL
);


CREATE TABLE RECOMMENDED_CAMPUS (
  RECOMMENDATION_ID INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
  CAMPUS_NAME VARCHAR(30) NOT NULL,
  CAMPUS_LOCATION VARCHAR(50) NOT NULL
);

DELIMITER //
CREATE PROCEDURE InsertCandidates()
BEGIN

/*Variables*/
SET @i = 1;
SET @maxRows = 700;

/*Loop to insert rows*/
WHILE @i <= @maxRows
DO
  SET @fname = CONCAT('FNAME', @i);
  SET @lname = CONCAT('LNAME', @i);
  SET @gender = CASE WHEN RAND() > 0.73 THEN 'F' ELSE 'M' END;
  SET @departmentId = CEIL(RAND() * 5);
  SET @appliedPosition = CASE
        WHEN RAND() < 0.3 THEN 'ANALYST'
        WHEN RAND() < 0.5 THEN 'ASSOCIATE'
        WHEN RAND() < 0.7 THEN 'VP'
        WHEN RAND() < 0.8 THEN 'ED'
        ELSE 'MD'
    END;
  SET @appliedThrough = CASE
        WHEN RAND() < 0.25 THEN 'CAMPUS'
        WHEN RAND() < 0.5 THEN 'VENDOR'
        WHEN RAND() < 0.75 THEN 'REFERENCE'
        ELSE 'WEBSITE'
    END;
  SET @status = CASE 
        WHEN RAND() < 0.33 THEN 'STAGE 1' 
        WHEN RAND() < 0.66 THEN 'STAGE 2'
        ELSE 'HIRED'
        END;
  SET @experienceYears = CEIL(RAND() * 20);
  set @vendorId = CASE
	WHEN @appliedThrough = 'VENDOR' THEN CEIL(RAND() * 4)
	 
	ELSE NULL
    END;
  set @campusId = CASE
	WHEN @appliedThrough = 'CAMPUS' THEN CEIL(RAND() * 4)
	ELSE NULL
    END;
  set @applicationDate = DATE_ADD('2016-01-01', INTERVAL CEIL(RAND() * 2737) DAY);

 /*Insert query*/
  INSERT INTO candidate (CANDIDATE_FNAME, CANDIDATE_LNAME, GENDER, DEPARTMENT_ID, APPLIED_POSITION, APPLIED_THROUGH, STATUS, EXPERIENCE_YEARS, VENDOR_ID, CAMPUS_ID, APPLICATION_DATE)
  VALUES (@fname, @lname, @gender, @departmentId, @appliedPosition, @appliedThrough, @status, @experienceYears, @vendorId, @campusId, @applicationDate);

  /*Increment counter*/
  SET @i = @i + 1;
END WHILE;
END//

DELIMITER ;

CALL InsertCandidates();

DELIMITER //
CREATE PROCEDURE InsertRoles()
BEGIN
SET @i = 1;
SET @maxRows = 150;
WHILE @i <= @maxRows
DO
SET @departmentId = CEIL(RAND() * 5);
SET @position = CASE
        WHEN RAND() < 0.3 THEN 'ANALYST'
        WHEN RAND() < 0.5 THEN 'ASSOCIATE'
        WHEN RAND() < 0.7 THEN 'VP'
        WHEN RAND() < 0.8 THEN 'ED'
        ELSE 'MD'
    END;
SET @bu = CASE
        WHEN RAND() < 0.25 THEN 'MANUFACTURING'
        WHEN RAND() < 0.5 THEN 'HEALTHCARE'
        WHEN RAND() < 0.75 THEN 'BANKING'
        ELSE 'LOGISTICS'
END;
SET @creationDate = DATE_ADD('2016-01-01', INTERVAL CEIL(RAND() * 2737) DAY);
INSERT INTO OPEN_ROLES (DEPARTMENT_ID, POSITION, BUSINESS_UNIT, CREATION_DATE)
VALUES(@departmentId,@position,@bu,@creationDate);
SET @i = @i + 1;
END WHILE;
END//

DELIMITER ;

CALL InsertRoles();

DELIMITER //
CREATE PROCEDURE InsertPerformance()
BEGIN

DECLARE finished INTEGER DEFAULT 0;
DECLARE eid INT;

DECLARE cur CURSOR FOR SELECT EMPLOYEE_ID FROM EMPLOYEE;
DECLARE CONTINUE HANDLER FOR NOT FOUND SET finished = 1; 


OPEN cur;

getEID: LOOP
FETCH cur INTO eid;
IF finished = 1 
THEN LEAVE getEID;
END IF;
SET @scoreYear = (SELECT YEAR(HIRING_DATE) FROM EMPLOYEE WHERE EMPLOYEE_ID = eid);

SET @endYear = YEAR(CURDATE());
WHILE @scoreYear < @endYear
DO
INSERT INTO PERFORMANCE (EMPLOYEE_ID, SCORE_YEAR, SCORE)
VALUES (eid, @scoreYear, CEIL(RAND() * 5));
SET @scoreYear = @scoreYear + 1;
END WHILE;
END LOOP getEID;
CLOSE cur;
END//
DELIMITER ;

CALL InsertPerformance();
